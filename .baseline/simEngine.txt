// FUNCTION_START simEngine — PROTECTED, baseline in .baseline/simEngine.txt
/* HALO simulation — engine
 * Generates synthetic days (outings, wandering, calendar events), simulates GPS,
 * runs four detectors side by side, and scores them.
 *
 * Detectors
 *   static   : alert whenever the tag leaves the home geofence
 *   routine  : home geofence + allowances learned from 4 weeks of location history
 *              (a stand-in for commercial "learned routine" products)
 *   mozo     : home geofence + Mobile Zones built from confirmed calendar events
 *   hybrid   : mozo OR routine
 */
(function (root) {
  var H = root.HALO, Wd = H.world, R = H.rng;
  var DAY = 1440, DRIVE = 450, WALK = 55; // meters per minute
  var DOW = ['Sunday', 'Monday', 'Tuesday', 'Wednesday', 'Thursday', 'Friday', 'Saturday'];
  var DETECTORS = [
    { key: 'static',  name: 'Static geofence' },
    { key: 'routine', name: 'Learned routine' },
    { key: 'mozo',    name: 'MoZo' },
    { key: 'hybrid',  name: 'MoZo + routine' }
  ];

  /* Weekly habits (dow: 0 = Sunday). Each happens with 85% probability. */
  var HABITS = [
    { place: 'church',   dow: 0, start: 540, dur: 90 },
    { place: 'pharmacy', dow: 1, start: 900, dur: 20 },
    { place: 'grocery',  dow: 2, start: 600, dur: 45 },
    { place: 'center',   dow: 4, start: 780, dur: 120 },
    { place: 'grocery',  dow: 6, start: 630, dur: 45 }
  ];
  var IRREGULAR = ['clinic', 'friend', 'diner', 'bank'];

  var DEFAULTS = {
    seed: 2026,
    days: 168,             // 24 weeks
    assistant: 'ask',      // 'ask' | 'assume'
    gate: true,            // caregiver approval for new events
    W: 150,                // corridor half-width (m)
    buf: 20,               // time buffer around events (min)
    R: 150,                // home geofence radius (m)
    sigma: 15,             // GPS noise, 1 s.d. (m)
    dropout: 0.03,         // probability a GPS fix is missing
    detour: 0.15,          // probability a trip uses an alternative route
    errand: 0.25,          // probability of a quick stop just off the route
    venueWander: 0.05,     // probability she walks away from a venue during an outing
    wanderPerWeek: 2,
    spuriousPerWeek: 1,    // hallucinated or injected events reaching the pipeline
    captureHabit: 0.6,     // share of habitual outings mentioned in conversation
    captureIrregular: 0.95,// share of one-off outings mentioned in conversation
    pComplete: 0.55,       // share of mentions that already contain every field
    extractErr: 0.07,      // LLM extraction error rate (≈ 1 − validation F1)
    readbackCatch: 0.8,    // share of wrong events caught when Halo reads them back
    gateCatch: 0.95        // share of spurious events rejected by the caregiver
  };

  function P(key) { return Wd.place(key); }
  function home() { return P('home'); }
  function uni(rng, a, b) { return a + rng() * (b - a); }
  function pick(rng, arr) { return arr[Math.floor(rng() * arr.length)]; }
  function poisson(rng, lam) { var L = Math.exp(-lam), k = 0, p = 1; do { k++; p *= rng(); } while (p > L); return k - 1; }
  function clampDay(m) { return Math.max(0, Math.min(DAY - 1, m)); }
  function dowOf(d) { return ((d % 7) + 7) % 7; }
  function fmtTime(m) {
    m = ((Math.round(m) % DAY) + DAY) % DAY;
    var h = Math.floor(m / 60), mm = m % 60, ap = h < 12 ? 'AM' : 'PM', h12 = h % 12 || 12;
    return h12 + ':' + (mm < 10 ? '0' : '') + mm + ' ' + ap;
  }

  /* ---------- outings ---------- */
  function planOutings(rng, d) {
    var dow = dowOf(d), outs = [];
    HABITS.forEach(function (h) {
      if (h.dow === dow && rng() < 0.85)
        outs.push({ place: h.place, kind: 'habit', start: Math.round(h.start + R.gauss(rng) * 10), dur: h.dur });
    });
    if (rng() < 0.3) {
      var o = { place: pick(rng, IRREGULAR), kind: 'irregular', start: 540 + 15 * Math.floor(rng() * 33), dur: 15 * Math.round(uni(rng, 3, 8)) };
      if (!outs.some(function (x) { return Math.abs(x.start - o.start) < 240; })) outs.push(o);
    }
    return outs;
  }
  /* Per-minute positions from a list of legs: {move: poly} or {stay: pos, n: minutes}. */
  function legsTrack(legs, speed) {
    var out = [];
    legs.forEach(function (leg) {
      if (leg.stay) { for (var i = 0; i < leg.n; i++) out.push(leg.stay); return; }
      var L = Wd.polyLength(leg.move), n = Math.ceil(L / speed);
      for (var j = 0; j < n; j++) out.push(Wd.pointAt(leg.move, j * speed));
    });
    return out;
  }
  function splitPoly(poly, d) {
    var pre = [poly[0]], acc = 0, i;
    for (i = 1; i < poly.length; i++) {
      var L = Wd.dist(poly[i - 1], poly[i]);
      if (acc + L >= d) break;
      acc += L; pre.push(poly[i]);
    }
    var cut = Wd.pointAt(poly, d);
    pre.push(cut);
    return { pre: pre, post: [cut].concat(poly.slice(i)), cut: cut };
  }
  function buildOuting(rng, o, p) {
    var h = home().node, dest = P(o.place), L, ang, off;
    o.detour = rng() < p.detour;
    o.route = o.detour ? Wd.detour(h, dest.node) : Wd.route(h, dest.node);
    L = o.route.length;
    ang = rng() * 2 * Math.PI; off = uni(rng, 20, 150); // parking lot / entrance is not exactly on the street corner
    o.stayPos = [dest.pos[0] + Math.cos(ang) * off, dest.pos[1] + Math.sin(ang) * off];
    var legs = [];
    o.errand = rng() < p.errand && L > 800;
    if (o.errand) { // quick stop just off the route (gas station, drive-through)
      var sp = splitPoly(o.route.poly, uni(rng, 0.3, 0.7) * L), c = sp.cut;
      var nxt = Wd.pointAt(o.route.poly, Math.min(L, Wd.polyLength(sp.pre) + 5));
      var dx = nxt[0] - c[0], dy = nxt[1] - c[1], n = Math.hypot(dx, dy) || 1, sgn = rng() < 0.5 ? -1 : 1, so = uni(rng, 150, 320);
      o.errandPos = [c[0] - dy / n * so * sgn, c[1] + dx / n * so * sgn];
      legs.push({ move: sp.pre }, { move: [c, o.errandPos] }, { stay: o.errandPos, n: Math.round(uni(rng, 3, 8)) }, { move: [o.errandPos, c] }, { move: sp.post });
    } else legs.push({ move: o.route.poly });
    legs.push({ move: [dest.pos, o.stayPos] });
    o.outTrack = legsTrack(legs, DRIVE);
    o.backTrack = legsTrack([{ move: [o.stayPos, dest.pos] }, { move: o.route.poly.slice().reverse() }], DRIVE);
    o.travel = Math.ceil(L / DRIVE);
    o.depart = o.start - o.outTrack.length - Math.round(uni(rng, 3, 12));
    o.arrive = o.depart + o.outTrack.length;
    o.actualEnd = o.start + o.dur + Math.round(uni(rng, -5, 25));
    o.back = o.actualEnd + o.backTrack.length;
    return o;
  }

  /* ---------- wandering ---------- */
  var PATTERNS = ['direct', 'random', 'lapping', 'pacing'];
  function wanderStart(rng) {
    var u = rng();
    if (u < 0.25) return rng() < 0.6 ? Math.round(uni(rng, 1260, 1380)) : Math.round(uni(rng, 0, 150)); // night
    if (u < 0.6) return Math.round(uni(rng, 900, 1140));  // late afternoon ("sundowning")
    return Math.round(uni(rng, 480, 900));                 // daytime
  }
  function extendRandom(rng, seq, n) {
    for (var i = 0; i < n; i++) {
      var cur = seq[seq.length - 1], prev = seq[seq.length - 2];
      var nb = Wd.neighbors(cur).filter(function (x) { return x !== prev; });
      seq.push(pick(rng, nb));
    }
    return seq;
  }
  function wanderNodes(rng, pattern) {
    var h = home().node, c = Wd.xy(h), hx = c[0], hy = c[1], seq = [h], i;
    if (pattern === 'direct') {
      var t; do { t = Math.floor(rng() * Wd.NX * Wd.NY); } while (Wd.manhattan(t, h) < 4);
      seq = extendRandom(rng, Wd.walkPath(h, t), 10);
    } else if (pattern === 'random') {
      seq = extendRandom(rng, seq, 18);
    } else if (pattern === 'lapping') {
      for (i = 0; i < 6; i++) seq.push(Wd.id(hx + 1, hy), Wd.id(hx + 1, hy - 1), Wd.id(hx, hy - 1), h);
    } else { // pacing
      for (i = 0; i < 8; i++) seq.push(Wd.id(hx + 2, hy), h);
    }
    return seq;
  }
  function planWanders(rng, outs, p) {
    var n = poisson(rng, p.wanderPerWeek / 7), list = [];
    for (var k = 0; k < n; k++) {
      for (var tries = 0; tries < 12; tries++) {
        var start = wanderStart(rng), dur = Math.round(uni(rng, 45, 110));
        var end = Math.min(DAY - 1, start + dur);
        var clash = outs.some(function (o) { return start < o.back + 2 && end > o.depart - 2; }) ||
                    list.some(function (w) { return start < w.end + 30 && end > w.start - 30; });
        if (clash || end - start < 30) continue;
        var pattern = pick(rng, PATTERNS);
        list.push({ start: start, end: end, pattern: pattern, origin: home().pos, poly: Wd.makeRoute(wanderNodes(rng, pattern)).poly });
        break;
      }
    }
    return list;
  }

  /* Walking away from a venue partway through an outing (e.g. leaving the day program). */
  function planVenueWanders(rng, outs, p) {
    var list = [];
    outs.forEach(function (o) {
      var stay = o.actualEnd - o.arrive;
      if (stay < 30 || rng() >= p.venueWander) return;
      var start = o.arrive + Math.round(uni(rng, 10, stay - 10)), dur = Math.round(uni(rng, 30, 70));
      var nodes = extendRandom(rng, [P(o.place).node], 10);
      var w = { start: start, end: Math.min(DAY - 1, start + dur), pattern: 'from ' + P(o.place).short, venue: o.place, origin: o.stayPos,
                poly: [o.stayPos].concat(Wd.makeRoute(nodes).poly) };
      o.actualEnd = Math.max(o.actualEnd, w.end + 5);
      o.back = o.actualEnd + o.backTrack.length;
      list.push(w);
    });
    return list;
  }

  /* ---------- the Synthetic Memory model (conversation -> calendar) ---------- */
  function calendarFor(rng, outs, spurious, p) {
    var ev = [];
    outs.forEach(function (o) {
      o.captured = rng() < (o.kind === 'habit' ? p.captureHabit : p.captureIrregular);
      if (!o.captured) return;
      var e = { place: o.place, start: o.start, end: o.start + o.dur, outing: o, issue: null };
      var complete = rng() < p.pComplete;
      if (!complete && p.assistant === 'assume') {
        var miss = pick(rng, ['time', 'place', 'duration']);
        if (miss === 'time' && o.start !== 540) { e.start = 540; e.end = 540 + o.dur; e.issue = 'Assumed 9:00 AM start'; }
        else if (miss === 'place') { e.place = null; e.issue = 'No location saved'; }
        else if (miss === 'duration' && o.dur > 60) { e.end = e.start + 60; e.issue = 'Assumed 1 hour'; }
      }
      if (rng() < p.extractErr) {
        var shift = (rng() < 0.5 ? -1 : 1) * 60 * (1 + Math.floor(rng() * 3));
        e.start += shift; e.end += shift;
        e.issue = (e.issue ? e.issue + '; ' : '') + 'Extraction error (' + (shift > 0 ? '+' : '') + shift / 60 + ' h)';
      }
      if (p.assistant === 'ask' && e.issue && rng() < p.readbackCatch) {
        e.caught = e.issue; e.issue = null;
        e.place = o.place; e.start = o.start; e.end = o.start + o.dur;
      }
      e.start = clampDay(e.start); e.end = clampDay(e.end);
      ev.push(e);
    });
    spurious.forEach(function (s) {
      s.blocked = p.gate && rng() < p.gateCatch;
      if (!s.blocked) ev.push({ place: s.place, start: s.start, end: s.start + s.dur, spurious: true, issue: 'Spurious event (never said by Margaret)' });
    });
    return ev;
  }
  function planSpurious(rng, p) {
    var n = poisson(rng, p.spuriousPerWeek / 7), list = [];
    var choices = Wd.PLACES.filter(function (x) { return x.key !== 'home'; }).map(function (x) { return x.key; });
    for (var i = 0; i < n; i++) list.push({ place: pick(rng, choices), start: 15 * Math.round(uni(rng, 600, 1320) / 15), dur: 60 });
    return list;
  }

  /* ---------- allowances (where and when leaving home is expected) ---------- */
  function allowance(placeKey, from, to, label, kind) {
    var r = Wd.route(home().node, P(placeKey).node);
    return { place: placeKey, from: from, to: to, poly: r.poly, dest: P(placeKey).pos, label: label, kind: kind };
  }
  function mozoAllowances(events, p) {
    return events.filter(function (e) { return e.place; }).map(function (e) {
      var tt = Math.ceil(Wd.route(home().node, P(e.place).node).length / DRIVE);
      var a = allowance(e.place, e.start - tt - p.buf, e.end + tt + p.buf, P(e.place).short, 'mozo');
      a.event = e; return a;
    });
  }
  function learnRoutine(p) {
    var rng = R.mulberry32(p.seed * 7 + 13), trips = {};
    for (var d = -28; d < 0; d++) {
      planOutings(rng, d).forEach(function (o) {
        buildOuting(rng, o, p);
        var k = o.place + '|' + dowOf(d);
        (trips[k] = trips[k] || []).push(o);
      });
    }
    var habits = [];
    Object.keys(trips).forEach(function (k) {
      var list = trips[k];
      if (list.length < 3) return;
      habits.push({
        place: list[0].place, dow: +k.split('|')[1],
        from: Math.min.apply(null, list.map(function (o) { return o.depart; })),
        to: Math.max.apply(null, list.map(function (o) { return o.back; })),
        seen: list.length
      });
    });
    return habits;
  }
  function inAllowance(a, m, q, W) {
    if (m < a.from || m > a.to) return false;
    if (Wd.distToPoly(q, a.poly) <= W) return true;
    return Wd.dist(q, a.dest) <= W + 60;
  }

  /* ---------- one day ---------- */
  function simulateDay(d, p, habits) {
    var rng = R.mulberry32((p.seed * 1000003 + d * 7919) >>> 0);
    var outs = planOutings(rng, d);
    outs.forEach(function (o) { buildOuting(rng, o, p); });
    var venueW = planVenueWanders(rng, outs, p);
    var wanders = planWanders(rng, outs, p).concat(venueW).sort(function (a, b) { return a.start - b.start; });
    var spurious = planSpurious(rng, p);
    var events = calendarFor(rng, outs, spurious, p);

    var hp = home().pos, tx = new Float32Array(DAY), ty = new Float32Array(DAY), ox = new Float32Array(DAY), oy = new Float32Array(DAY);
    var away = new Uint8Array(DAY), m, q;
    for (m = 0; m < DAY; m++) { tx[m] = hp[0] + R.gauss(rng) * 8; ty[m] = hp[1] + R.gauss(rng) * 8; }
    outs.forEach(function (o) {
      for (m = Math.max(0, o.depart); m < Math.min(DAY, o.back); m++) {
        if (m < o.arrive) q = o.outTrack[m - o.depart];
        else if (m < o.actualEnd) q = [o.stayPos[0] + R.gauss(rng) * 10, o.stayPos[1] + R.gauss(rng) * 10];
        else q = o.backTrack[Math.min(o.backTrack.length - 1, m - o.actualEnd)];
        tx[m] = q[0]; ty[m] = q[1]; away[m] = 1;
      }
    });
    wanders.forEach(function (w) {
      w.firstExit = -1;
      for (m = w.start; m < w.end; m++) {
        q = Wd.pointAt(w.poly, (m - w.start) * WALK);
        tx[m] = q[0]; ty[m] = q[1];
        if (w.firstExit < 0 && Wd.dist(q, w.origin) > (w.venue ? 150 : p.R)) w.firstExit = m;
      }
      w.leftHome = w.firstExit >= 0;
    });
    for (m = 0; m < DAY; m++) {
      if (rng() < p.dropout) { ox[m] = NaN; oy[m] = NaN; }
      else { ox[m] = tx[m] + R.gauss(rng) * p.sigma; oy[m] = ty[m] + R.gauss(rng) * p.sigma; }
    }

    var day = { d: d, dow: dowOf(d), outs: outs, wanders: wanders, spurious: spurious, events: events, tx: tx, ty: ty, ox: ox, oy: oy, away: away };
    day.mozoA = mozoAllowances(events, p);
    day.routA = habits.filter(function (h) { return h.dow === day.dow; }).map(function (h) {
      return allowance(h.place, h.from - p.buf, h.to + p.buf, P(h.place).short, 'routine');
    });
    evaluate(day, p);
    return day;
  }

  function evaluate(day, p) {
    var hp = home().pos, res = {};
    DETECTORS.forEach(function (D) { res[D.key] = { alarm: new Uint8Array(DAY), onsets: [], out: 0, inn: 0, on: false }; });
    for (var m = 0; m < DAY; m++) {
      if (isNaN(day.ox[m])) { DETECTORS.forEach(function (D) { res[D.key].alarm[m] = res[D.key].on ? 1 : 0; }); continue; }
      var q = [day.ox[m], day.oy[m]], atHome = Wd.dist(q, hp) <= p.R, mz = atHome, rt = atHome;
      if (!atHome) {
        mz = day.mozoA.some(function (a) { return inAllowance(a, m, q, p.W); });
        rt = day.routA.some(function (a) { return inAllowance(a, m, q, p.W); });
      }
      var ok = { 'static': atHome, routine: rt, mozo: mz, hybrid: mz || rt };
      DETECTORS.forEach(function (D) {
        var s = res[D.key];
        if (!ok[D.key]) { s.out++; s.inn = 0; if (!s.on && s.out >= 2) { s.on = true; s.onsets.push(m); } }
        else { s.inn++; s.out = 0; if (s.on && s.inn >= 10) s.on = false; }
        s.alarm[m] = s.on ? 1 : 0;
      });
    }
    var open = { 'static': 0, routine: 0, mozo: 0, hybrid: 0 };
    function anyOpen(list, t) { for (var i = 0; i < list.length; i++) if (t >= list[i].from && t <= list[i].to) return true; return false; }
    for (var t = 0; t < DAY; t++) {
      if (day.away[t]) continue;
      var a = anyOpen(day.mozoA, t), b = anyOpen(day.routA, t);
      if (a) open.mozo++; if (b) open.routine++; if (a || b) open.hybrid++;
    }
    var stats = {};
    DETECTORS.forEach(function (D) {
      var s = res[D.key], st = { fa: 0, ep: 0, det: 0, fast: 0, lat: [], falseOnsets: [], open: open[D.key] };
      s.onsets.forEach(function (t) {
        var real = day.wanders.some(function (w) { return t >= w.start && t <= w.end + 5; });
        if (!real) { st.fa++; st.falseOnsets.push(t); }
      });
      day.wanders.forEach(function (w) {
        if (!w.leftHome) return;
        st.ep++;
        var hit = s.alarm[w.firstExit] ? w.firstExit : -1;
        if (hit < 0) for (var i = 0; i < s.onsets.length; i++) if (s.onsets[i] >= w.start && s.onsets[i] <= w.end + 5) { hit = s.onsets[i]; break; }
        w[D.key] = hit < 0 ? null : Math.max(0, hit - w.firstExit);
        if (hit >= 0) { st.det++; st.lat.push(Math.max(0, hit - w.firstExit)); if (hit - w.firstExit <= 5) st.fast++; }
      });
      stats[D.key] = st;
      delete s.out; delete s.inn; delete s.on;
    });
    day.res = res; day.stats = stats;
  }

  /* ---------- batch + statistics ---------- */
  function median(a) { if (!a.length) return NaN; var s = a.slice().sort(function (x, y) { return x - y; }), k = s.length >> 1; return s.length % 2 ? s[k] : (s[k - 1] + s[k]) / 2; }
  function pct(sorted, q) { if (!sorted.length) return NaN; var i = Math.min(sorted.length - 1, Math.max(0, Math.round(q * (sorted.length - 1)))); return sorted[i]; }

  function summarize(days, seed) {
    var N = days.length, rng = R.mulberry32(seed + 99), B = 400, out = {};
    DETECTORS.forEach(function (D) {
      var k = D.key, fa = 0, ep = 0, det = 0, fast = 0, open = 0, lat = [];
      days.forEach(function (d) { var s = d.stats[k]; fa += s.fa; ep += s.ep; det += s.det; fast += s.fast; open += s.open; lat = lat.concat(s.lat); });
      var bF = [], bD = [];
      for (var b = 0; b < B; b++) {
        var f = 0, e = 0, t = 0;
        for (var i = 0; i < N; i++) { var s = days[Math.floor(rng() * N)].stats[k]; f += s.fa; e += s.ep; t += s.det; }
        bF.push(7 * f / N); if (e) bD.push(t / e);
      }
      bF.sort(function (x, y) { return x - y; }); bD.sort(function (x, y) { return x - y; });
      out[k] = {
        faWeek: 7 * fa / N, faLo: pct(bF, 0.025), faHi: pct(bF, 0.975),
        detRate: ep ? det / ep : NaN, detLo: pct(bD, 0.025), detHi: pct(bD, 0.975),
        latMed: median(lat), latMean: lat.length ? lat.reduce(function (a, b) { return a + b; }, 0) / lat.length : NaN, fastRate: ep ? fast / ep : NaN, openHoursWeek: 7 * open / N / 60,
        episodes: ep, detected: det, falseAlarms: fa
      };
    });
    return out;
  }

  function run(params) {
    var p = Object.assign({}, DEFAULTS, params || {});
    var habits = learnRoutine(p), days = [];
    for (var d = 0; d < p.days; d++) days.push(simulateDay(d, p, habits));
    var totals = { venueWanders: 0, errands: 0, outings: 0, irregular: 0, wanders: 0, nearHome: 0, spurious: 0, spuriousApplied: 0, events: 0, eventIssues: 0, caught: 0, detours: 0 };
    days.forEach(function (day) {
      totals.outings += day.outs.length;
      totals.irregular += day.outs.filter(function (o) { return o.kind === 'irregular'; }).length;
      totals.detours += day.outs.filter(function (o) { return o.detour; }).length;
      totals.wanders += day.wanders.length;
      totals.venueWanders += day.wanders.filter(function (w) { return w.venue; }).length;
      totals.errands += day.outs.filter(function (o) { return o.errand; }).length;
      totals.nearHome += day.wanders.filter(function (w) { return !w.leftHome; }).length;
      totals.spurious += day.spurious.length;
      totals.spuriousApplied += day.events.filter(function (e) { return e.spurious; }).length;
      totals.events += day.events.length;
      totals.eventIssues += day.events.filter(function (e) { return e.issue; }).length;
      totals.caught += day.events.filter(function (e) { return e.caught; }).length;
    });
    return { params: p, habits: habits, days: days, summary: summarize(days, p.seed), totals: totals };
  }
